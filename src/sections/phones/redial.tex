\section{Redialing Numbers}

\subsubsection{To redial the last dialed number from your phone:}

\begin{enumerate}
	\item Press \scalerel*{\includegraphics{images/yealink/btn-rd.png}}{B} twice.

		A call to your last dialed number is attempted.
\end{enumerate}

\subsubsection{To redial a previously dialed number from your phone:}

\begin{enumerate}
	\item Press \scalerel*{\includegraphics{images/yealink/btn-rd.png}}{B} when
		the phone is idle.
	\item Press \scalerel*{\includegraphics{images/yealink/btn-up.png}}{B} or
		\scalerel*{\includegraphics{images/yealink/btn-down.png}}{B} to select
		the desired entry from the placed calls list, and then press
		\scalerel*{\includegraphics{images/yealink/btn-rd.png}}{B} or the
		\textbf{Send} soft key.
\end{enumerate}
